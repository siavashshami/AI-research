import numpy as np
import pandas as pd
from sklearn import metrics
from sklearn.ensemble import RandomForestRegressor


# NOTICE EW and UD 
Train = pd.read_csv(r"2016.csv")
Test = pd.read_csv(r"2020.csv")

# Here We Read The 2016 for Train
X_train = Train.drop(['FID', 'FID_EW2016', 'FID_EWClip', 'pointid', 'Vel'],axis=1)
y_train = Train['Vel']

# Here We Read The 2020 for Test
X_test = Test.drop(['FID', 'FID_EW2020', 'FID_EWP202', 'pointid', 'Vel'],axis=1)
y_test = Test['Vel']


# define the model
Random_model = RandomForestRegressor(n_estimators=100)
# fit the model
Random_model.fit(X_train, y_train)
Random_pred = Random_model.predict(X_test)

RMSE_R = ('{:.2f}'.format(np.sqrt(metrics.mean_squared_error(y_test,Random_pred))))

MAE_R = ('{:.2f}'.format(metrics.mean_absolute_error(y_test,Random_pred)))

R2_R = ('{:.2f}'.format(metrics.r2_score(y_test,Random_pred)*100))
corr = np.corrcoef(y_test,Random_pred)

CO = corr[0][1]

R = ('{:.0f}'.format((CO)*100))

print(RMSE_R, R)

# Running SVR  

from sklearn.svm import SVR


# define the model
SVR_model = SVR(kernel='rbf', C=100)
# fit the model
SVR_model.fit(X_train, y_train)
SVR_pred = SVR_model.predict(X_test)

RMSE_S = ('{:.2f}'.format(np.sqrt(metrics.mean_squared_error(y_test,SVR_pred))))

MAE_S = ('{:.2f}'.format(metrics.mean_absolute_error(y_test,SVR_pred)))

R2_S = ('{:.2f}'.format(metrics.r2_score(y_test,SVR_pred)*100))
corr = np.corrcoef(y_test,SVR_pred)

CO = corr[0][1]

R = ('{:.0f}'.format((CO)*100))



# This Section Run Code for 100 times
AR_RF = []
AR2_RF = []
ARMSE_RF = []
AMAE_RF = []

for i in range(1,101):
    
    
    
    # Define The Model consider RFR and SVR be tried
    Random_model = RandomForestRegressor(n_estimators=100)
    # fit the model
    Random_model.fit(X_train, y_train)
    Random_pred = Random_model.predict(X_test)
    
    RMSE_R = ('{:.2f}'.format(np.sqrt(metrics.mean_squared_error(y_test,Random_pred))))
    
    MAE_R = ('{:.2f}'.format(metrics.mean_absolute_error(y_test,Random_pred)))
    
    R2_R = ('{:.2f}'.format(metrics.r2_score(y_test,Random_pred)*100))
    corr = np.corrcoef(y_test,Random_pred)
    
    CO = corr[0][1]
    
    R = ('{:.0f}'.format((CO)*100))
    
    AR_RF.append(CO)
    AR2_RF.append(R2_R)
    ARMSE_RF.append(RMSE_R)
    AMAE_RF.append(MAE_R)
    print(i)
