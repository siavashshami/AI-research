
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import statistics

# Load the data from the CSV file into a pandas DataFrame
df = pd.read_csv(r"Histo.csv")

# Extract the column of interest from the DataFrame
data = df['UD2020']

# Create a histogram
plt.hist(data, bins=100, density=True, alpha=0.5, color='mediumturquoise',edgecolor='white')

# Create a distribution line
mu, sigma = np.mean(data), np.std(data)
x = np.linspace(mu - 3*sigma, mu + 3*sigma, 100)
y = (1/(sigma * np.sqrt(2 * np.pi))) * np.exp(-0.5 * ((x - mu) / sigma) ** 2)
plt.plot(x, y, color='darkblue')

plt.xlabel('Range Value',fontsize=15)
plt.ylabel('Frequnecy',fontsize=15)
plt.text(-48,0.354, 'Mean: -0.85\nVAR: 3.35\nMax: 3\nMin: -17', style='italic',fontsize=15, bbox={
        'facecolor': 'whitesmoke', 'alpha': 1, 'pad': 10})
plt.xlim(-50,50)
plt.axvline(x = 0, color = 'black', linewidth=1.5,linestyle='--')

plt.tick_params(axis="x", labelsize=15)
plt.tick_params(axis='y', labelsize=15)


plt.grid(linestyle='--')
# Show the plot
plt.show()





print('Mean: ',sum(df['UD2020'])/len(df['UD2020']))
print('Var: ',statistics.variance(df['UD2020']))
print('Min',min(df['UD2020']))
print('Max: ',max(df['UD2020']))
