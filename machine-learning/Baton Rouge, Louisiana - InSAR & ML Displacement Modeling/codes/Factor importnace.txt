
import pandas as pd
import matplotlib.pyplot as plt

plt.rcParams['savefig.dpi'] = 500

df = pd.DataFrame({'KNN': [0.057,0.021,0.41,0.088,0.008,0.21,0.19], 'GBR': [0.047,0.004,0.49,0.019,0.013,0.41,0.009],
                   'RFR': [0.028,0.001,0.53,0.005,0.011,0.41,0.006]}, 
                   index=['DEM','Slope','Fault','River','Road','PRC','LU'])

ax = df.plot(kind='bar', width=0.8,color=['aquamarine', 'lightseagreen','deepskyblue']) 
x_offset = 0
y_offset = 0.0
for p in ax.patches:
    b = p.get_bbox()
    val = "{:.3f}".format(b.y1 + b.y0)        
    ax.annotate(val, ((b.x0 + b.x1)/2 + x_offset, b.y1 + y_offset),ha = 'center',fontsize=13,bbox = dict(facecolor = 'white',
                edgecolor='black'))
    
    
    
    
    
plt.rcParams["font.family"] = "Times New Roman"

plt.ylabel('Factor Importance',fontsize=30)

plt.tick_params(axis="y", labelsize=30)
plt.tick_params(axis="x", labelsize=30,rotation=45)

plt.tick_params(axis="y", labelsize=30)
plt.tick_params(axis="x", labelsize=30,rotation=45)
plt.ylim(0,0.6)
plt.legend(loc ="upper left",fontsize="xx-large",shadow=True)

plt.grid(axis='y',linestyle = '--')
