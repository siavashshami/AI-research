# ================================  First ================================

plt.subplot(1, 3, 1)  



def addlabels(x,y):
    for i in range(len(x)):
        plt.text(i, y[i], y[i], ha = 'center', fontsize=20,
                 bbox = dict(facecolor = 'white', alpha =.8))
 
if __name__ == '__main__':
    R2 = {'LASSO':0, 'KNN':0.84, 'GBR':0.85,
        'RFR':0.77}


    courses = list(R2.keys())
    values = list(R2.values())
      
     
    # creating the bar plot
    plt.bar(courses, values, color ='aquamarine',
            width = 0.4)
    plt.rcParams["font.family"] = "Times New Roman"
    
    plt.ylim(0,1)
    
    plt.tick_params(axis="y", labelsize=25)
    plt.tick_params(axis="x", rotation=45,labelsize=25)
    
    
    plt.title('R2 (%)',fontsize=35)
    plt.grid(axis='y',color = 'gray', linestyle = '--', linewidth = 0.5) 

    # calling the function to add value labels
    addlabels(courses, values)

     
    # visualizing the plot
    plt.show()
